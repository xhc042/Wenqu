## 二、P0 任务详细说明

### P0-① 抽离 `_run_speed_mode_postprocess` 函数

#### 背景与目标

**背景**：app.py 中 speed 模式的处理逻辑在两处几乎逐行重复：
- `run_chapter_generation`（异步任务路径）：L261-333
- `generate_chapters`（同步 API 路径）：L731-839

两段都执行：循环 `extract_chapter_snapshot` → `generate_global_highlights` → `db.add_global_highlights` → `generate_speed_read_syllabus` → DELETE + INSERT syllabus_items。

**目标**：抽出一个独立的工具函数，让两处调用同一份逻辑，消除重复。

#### 代码位置

| 文件 | 行号 | 说明 |
|------|------|------|
| `app.py` | 261-333 | 异步任务路径中的 speed 处理 |
| `app.py` | 731-839 | 同步 API 路径中的 speed 处理 |

**Step 1**：在 `app.py` 顶部添加 helper（紧跟 `_add_default_syllabus_items`）：

```python
async def _run_speed_mode_postprocess(
    course_id: str,
    chapters,
    chapter_titles,
    source_type,
    source_path,
    concurrency: int = 3,
):
    """
    speed 模式后处理：快照生成 + 全局精华 + 精华掌握项。

    输入：分章结果（来自 smart_chunk），原 EPUB 路径（用于按需懒加载）
    输出：处理结果 dict（不会主动写 DB，由调用方决定）
    """
    import asyncio
    import logging
    logger = logging.getLogger(__name__)

    # --- Step 1: 准备 chapter_idx/content/title ---
    chapter_data = []  # [(chapter_idx, title, content)]
    for chapter_item in chapters:
        if isinstance(chapter_item, tuple) and len(chapter_item) >= 3:
            title, content, meta = chapter_item
            chapter_idx = meta.get("idx", 0)
        elif isinstance(chapter_item, tuple) and len(chapter_item) >= 2:
            title, content = chapter_item
            chapter_idx = 0
        else:
            continue
        chapter_data.append((chapter_idx, title, content))

    # --- Step 2: 并发生成快照（P0-② 的实现） ---
    sem = asyncio.Semaphore(concurrency)

    async def _gen_snapshot(ch_idx, title, content):
        async with sem:
            # speed 模式从 EPUB 提取完整内容用于快照（不限于首尾段）
            snapshot_content = content
            if source_type == "epub" and source_path:
                try:
                    from chunker import extract_toc_from_epub, extract_chapter_content_by_href
                    toc_items, _ = await extract_toc_from_epub(source_path)
                    for toc_item in toc_items:
                        if toc_item.get("idx") == ch_idx:
                            href = toc_item.get("href", "")
                            if href:
                                snapshot_content = await extract_chapter_content_by_href(
                                    source_path, href
                                )
                            break
                except Exception as e:
                    logger.warning(f"[speed] EPUB 内容提取失败 [{title[:20]}]: {e}")

            if not snapshot_content:
                return ch_idx, title, None

            try:
                snapshot = await extract_chapter_snapshot(snapshot_content, title)
                return ch_idx, title, snapshot
            except Exception as e:
                logger.warning(f"[speed] 快照生成失败 [{title[:20]}]: {e}")
                return ch_idx, title, None

    tasks = [_gen_snapshot(idx, title, content) for idx, title, content in chapter_data]
    results = await asyncio.gather(*tasks, return_exceptions=True)

    snapshots = []
    snapshot_by_idx = {}
    for r in results:
        if isinstance(r, Exception):
            logger.warning(f"[speed] gather 异常: {r}")
            continue
        ch_idx, title, snapshot = r
        if snapshot:
            snapshots.append(snapshot)
            snapshot_by_idx[ch_idx] = snapshot

    # --- Step 3: 全局精华 ---
    highlights = {}
    if snapshots:
        try:
            highlights = await generate_global_highlights(course_id, snapshots, chapter_titles)
        except Exception as e:
            logger.warning(f"[speed] 全局精华生成失败: {e}")
            highlights = {}

    # --- Step 4: 精华掌握项（P1-① 方案 B：直接用 key_points） ---
    syllabus_items = []
    key_points = highlights.get("key_points", []) if isinstance(highlights, dict) else []

    if key_points:
        # 把 key_points 分配到对应章节（按 snapshot 的 importance 排序）
        sorted_snaps = sorted(
            snapshot_by_idx.items(),
            key=lambda kv: kv[1].get("importance", 3),
            reverse=True,
        )
        # 简单的 round-robin 分配
        for i, kp in enumerate(key_points[:15]):  # 最多 15 条
            if i < len(sorted_snaps):
                target_ch_idx = sorted_snaps[i % len(sorted_snaps)][0]
            else:
                target_ch_idx = 0
            syllabus_items.append((target_ch_idx, kp))

    # --- Step 5: fallback（P1-④）---
    if not syllabus_items and snapshots:
        # 从每个快照的 learning_goal 兜底
        for ch_idx, snap in snapshot_by_idx.items():
            goal = snap.get("learning_goal", "").strip()
            if goal:
                syllabus_items.append((ch_idx, goal))
            if len(syllabus_items) >= 10:
                break

    return {
        "snapshots": snapshots,
        "snapshots_by_idx": snapshot_by_idx,
        "highlights": highlights,
        "syllabus_items": syllabus_items,
    }


async def _persist_speed_results(course_id: str, result: dict) -> None:
    """把后处理结果写入数据库（独立函数便于测试）"""
    import json
    import logging
    logger = logging.getLogger(__name__)

    snapshots_by_idx = result.get("snapshots_by_idx", {})

    # 1. 写 chapter_snapshots
    for ch_idx, snapshot in snapshots_by_idx.items():
        try:
            db.add_chapter_snapshot(
                course_id, ch_idx,
                json.dumps(snapshot.get("keywords", []), ensure_ascii=False),
                snapshot.get("core_viewpoint", ""),
                global_priority=snapshot.get("importance", 0),
                learning_goal=snapshot.get("learning_goal", ""),
                importance=snapshot.get("importance", 0),
                difficulty=snapshot.get("difficulty", ""),
            )
        except Exception as e:
            logger.warning(f"[speed] 写 chapter_snapshot 失败 [{ch_idx}]: {e}")

    # 2. 写 global_highlights
    highlights = result.get("highlights", {})
    if highlights:
        try:
            db.add_global_highlights(
                course_id,
                json.dumps(highlights.get("key_points", []), ensure_ascii=False),
                json.dumps(highlights.get("chapter_priorities", []), ensure_ascii=False),
                highlights.get("relationships", ""),
                json.dumps(highlights.get("core_chapter_indices", []), ensure_ascii=False),
                json.dumps(highlights.get("chapter_dependencies", {}), ensure_ascii=False),
            )
        except Exception as e:
            logger.warning(f"[speed] 写 global_highlights 失败: {e}")

    # 3. 删旧 syllabus_items + 写新的
    syllabus_items = result.get("syllabus_items", [])
    if syllabus_items:
        try:
            conn = db.get_conn()
            try:
                conn.execute("DELETE FROM syllabus_items WHERE course_id=?", (course_id,))
                conn.commit()
            finally:
                conn.close()
            for ch_idx, desc in syllabus_items:
                db.add_syllabus_item(course_id, ch_idx, desc)
        except Exception as e:
            logger.warning(f"[speed] 写 syllabus_items 失败: {e}")
```

### P0-② speed 快照生成改 `asyncio.gather` 并发

#### 背景与目标

**背景**：当前 speed 模式的快照生成是 `for await` 串行循环，50 章的书要等 50×3s ≈ 150s。
**目标**：用 `asyncio.gather + Semaphore` 控制并发，长书前置时间降低 60%+。

#### 代码位置

- `chunker.py` `extract_chapter_snapshot` 函数（L835-951）：单个快照的 LLM 调用
- `app.py` L266-294, L736-781：串行调用循环

#### 详细设计

**并发度选择**：3（实测经验，DeepSeek API rate limit 在 3 并发下稳定）
**限流方式**：`asyncio.Semaphore(3)`
**顺序保证**：用 `(chapter_idx, snapshot)` 元组，gather 后按 idx 排序

#### 实现步骤

**Step 1**：在 `chunker.py` 中新增批量接口（不破坏现有函数）：

```python
# chunker.py 新增（在 extract_chapter_snapshot 后）
async def extract_chapter_snapshots_batch(
    chapters: List[Tuple[int, str, str]],  # [(chapter_idx, title, content), ...]
    concurrency: int = 3,
) -> Dict[int, dict]:
    """
    并发生成多个章节的快照，返回 {chapter_idx: snapshot}

    Args:
        chapters: 待生成快照的章节列表
        concurrency: 最大并发数（默认 3，避免触发 API rate limit）
    """
    sem = asyncio.Semaphore(concurrency)

    async def _gen(idx: int, title: str, content: str):
        async with sem:
            try:
                snap = await extract_chapter_snapshot(content, title)
                return idx, snap
            except Exception as e:
                import logging
                logging.warning(f"批量快照生成失败 [{title[:20]}]: {e}")
                return idx, None

    tasks = [_gen(idx, title, content) for idx, title, content in chapters]
    results = await asyncio.gather(*tasks, return_exceptions=True)

    out = {}
    for r in results:
        if isinstance(r, Exception):
            continue
        idx, snap = r
        if snap:
            out[idx] = snap
    return out
```

## 三、P1 任务详细说明

### P1-① 用 `key_points` 替代 `generate_speed_read_syllabus`

#### 背景与目标

**背景**：`generate_speed_read_syllabus` 存在两个问题：
1. `chapter_info[:20]` 截断（chunker.py L1262），长书后 N 章不进 prompt
2. 第二次 LLM 调用冗余——`global_highlights.key_points` 已经是"全书 5-10 个核心知识点"

**目标**：直接用 `global_highlights.key_points` 作为精华掌握项来源，省一次 LLM 调用 + 修复长书截断。

#### 代码位置

- `chunker.py` L1222-1299：`generate_speed_read_syllabus`
- `app.py` L317-330, L799-812：调用点

#### 详细设计

**映射策略**：
- `key_points` 是"能+具体动作"描述（如 "能解释 ETF 运作机制"）
- 每个 key_point 需要分配一个 `chapter_index`
- 分配规则：按 snapshot 的 importance 排序，round-robin 分配

#### 实现步骤

**Step 1**：修改 `_run_speed_mode_postprocess`（已在 P0-① 中实现），关键代码：

```python
# 已在 P0-① 中实现

def _sample_first_last(text: str, max_per_sample: int = 300) -> str:
    """
    速读模式章节内容采样。
    - 多段：取首段 + 尾段
    - 单段：退化为均匀采样（避免信息丢失）

    paragraphs = [p.strip() for p in text.split('\n\n') if p.strip()]
    if not paragraphs:
        return text[:max_per_sample]

    if len(paragraphs) == 1:
        # 单段场景：整段塞进一个 <p>，用均匀采样保留头尾
        # 目标 max_per_sample * 2 字（首尾各 max_per_sample）
        sampled = sample_chapter_fairly(text, target_chars=max_per_sample * 2)
        return sampled

    first = paragraphs[0][:max_per_sample]
    last = paragraphs[-1][:max_per_sample]
    return f"{first}\n\n...（中间内容略）...\n\n{last}"

async def _check_debate_trigger(self, user_text: str) -> bool:
    """检测是否触发反方辩论"""
    # 第一步：否定词检测 — 如果有强否定词，不触发辩论
    negation_keywords = ["不对", "不同意", "但是", "可是", "然而", "我有不同看法", "不是这样的"]
    if any(kw in user_text for kw in negation_keywords):
        return False

    # 第二步：认同词计数
    agreement_keywords = ["是的", "对", "有道理", "没错", "你说得对", "确实", "同意", "明白了"]
    agreement_count = sum(1 for kw in agreement_keywords if kw in user_text)

    return agreement_count >= 2 and self.current_round > 2
```

### P3-② `_parse_chapter_num` 扩展正则

#### 代码位置

- `chunker.py` L1047-1049

#### 实现步骤

```python
# 修改后
import re

CN_NUM = {
    '一': 1, '二': 2, '三': 3, '四': 4, '五': 5,
    '六': 6, '七': 7, '八': 8, '九': 9, '十': 10,
}

def _parse_chapter_num(s: str) -> int:
    """
    解析章节号，支持：
    - 第N章 / 第N部分
    - 第五章（中文数字）
    - Chapter 5 / Chap. 5
    """
    s = str(s)
    # 阿拉伯数字
    m = re.search(r'第(\d+)[章部分]', s)
    if m:
        return int(m.group(1)) - 1
    # 中文数字
    m = re.search(r'第([一二三四五六七八九十]+)[章部分]', s)
    if m:
        cn = m.group(1)
        # 简单处理：十X = 10+X
        if cn == '十':
            return 10 - 1
        if cn.startswith('十'):
            return 10 + CN_NUM.get(cn[1:], 0) - 1
        return CN_NUM.get(cn, -1) - 1
    # 英文
    m = re.search(r'(?:Chapter|Chap\.?)\s*(\d+)', s, re.IGNORECASE)
    if m:
        return int(m.group(1)) - 1
    return -1
```

### P3-③ 心流检测多消息综合判断

#### 代码位置

- `state_machine.py` L394-402

#### 实现步骤

```python
# 修改后
def _check_flow_state(self, user_text: str):
    """心流检测：综合最近 3 条消息判断"""
    if self.current_round < FLOW_DETECTION["min_rounds"]:
        return

    # 当前消息判断
    current_signals = (
        len(user_text) > FLOW_DETECTION["reply_length_threshold"] and
        any(q in user_text for q in ["?", "为什么", "如何", "是不是"])
    )

    # 最近 3 条用户消息
    recent_user_msgs = [
        m for m in self.messages[-10:]
        if m.get("role") == "user"
    ][-3:]
    avg_len = sum(len(m.get("content", "")) for m in recent_user_msgs) / max(len(recent_user_msgs), 1)
    question_density = sum(
        1 for m in recent_user_msgs
        if any(q in m.get("content", "") for q in ["?", "为什么", "如何"])
    ) / max(len(recent_user_msgs), 1)

    if current_signals or (avg_len > 80 and question_density > 0.3):
        self.is_flow_state = True
        self.flow_extend_count += FLOW_DETECTION["auto_extend_rounds"]
    else:
        self.is_flow_state = False
```

## 一、需求概述

### 1.1 目标

基于现有三种学习模式（速读/细读/研读），解决以下核心问题：

| 模式 | 用户期望 | 现有实现 | 主要差距 |
|------|----------|----------|---------|
| 🚀 速读 | 快速学完全书，抓住最重要知识点 | 仅提取目录+首尾段，不生成掌握项，全部懒加载 | 缺乏知识点提炼、核心总结和快速验证机制 |
| 📖 细读 | 标准掌握全书知识点 | 全文前5000字，生成标准掌握项，全部预加载 | 掌握项列表与学习过程脱节，缺乏知识点覆盖度追踪和练习环节；内容截断导致后半本书丢失 |
| 🔬 研读 | 深度掌握全书知识点，辩证分析 | 保留全文，生成深度掌握项+苏格拉底追问，预加载3章 | 辩证分析依赖LLM自由发挥，缺乏结构化框架 |

### 1.2 优化原则

1. **速读抓核心**：通过"知识快照+全局精华"双轨制，让用户在30分钟内掌握全书最重要的5-10个知识点
2. **细读全覆盖**：通过"均匀采样+掌握项绑定+覆盖度追踪"，确保全书知识点不被遗漏
3. **研读能辩证**：通过"辩证框架+反方辩论+思辨笔记"，真正实现批判性思考
4. **模式联动契约**：速读/细读/研读与认知深度、学习时长智能联动
5. **目标导向时长**：学习时长决定学习目标而非对话轮次

def sample_chapter_fairly(text: str, target_chars: int = 5000) -> str:
    """
    按章节均匀采样，保证全书代表性
    不是只取前5000字，而是每章取固定比例
    """
    paragraphs = [p.strip() for p in text.split('\n\n') if p.strip()]
    if not paragraphs:
        return text[:target_chars]
    
    # 按比例分配每章的字符数
    chars_per_para = target_chars / len(paragraphs)
    sampled = []
    current_chars = 0
    
    for para in paragraphs:
        if current_chars >= target_chars:
            break
        sampled.append(para)
        current_chars += len(para)
    
    return '\n\n'.join(sampled)
```

def _check_knowledge_coverage(self) -> dict:
    """
    检查当前章节的知识点覆盖情况
    返回：哪些知识点已掌握、哪些还需要强化
    """
    chapter_items = [s for s in self.syllabus_items 
                     if s["chapter_index"] == self.chapter_index]
    
    coverage = {
        "total": len(chapter_items),
        "mastered": sum(1 for s in chapter_items if s["status"] == "mastered"),
        "in_progress": sum(1 for s in chapter_items if s["status"] == "in_progress"),
        "pending": sum(1 for s in chapter_items if s["status"] == "pending"),
    }
    
    # 如果本章还有pending项，提示用户继续学习
    if coverage["pending"] > 0 and coverage["mastered"] > 0:
        return {"partial": True, "suggestion": f"本章还有{coverage['pending']}个知识点未掌握"}
    
    return coverage
```

#### 4.3.2 反方辩论子模式（state_machine.py）

```python
# 新增状态常量
DEBATE = "DEBATE"

async def _check_debate_trigger(self, user_text: str) -> bool:
    """
    检测是否触发反方辩论
    条件：学生连续两次完全认同教师观点
    """
    agreement_keywords = ["是的", "对", "有道理", "没错", "你说得对", "确实"]
    agreement_count = sum(1 for kw in agreement_keywords if kw in user_text)
    
    # 如果认同词汇超过2个，且当前轮次>2，触发辩论
    return agreement_count >= 2 and self.current_round > 2

async def _start_debate(self) -> AsyncGenerator[str, None]:
    """切换到反方角色，提出对立观点"""
    self.state = self.DEBATE
    
    debate_prompt = f"""
    现在请你扮演与本章节观点对立的学者。
    针对刚才讨论的内容，提出一个有力的反对观点。
    
    要求：
    - 必须基于教材内容，不能凭空捏造
    - 观点要有学术依据，不能是情绪化反驳
    - 最后要求学生思考：哪种观点更有说服力？为什么？
    """
    
    self.messages.append({"role": "user", "content": debate_prompt})
    
    content_parts = []
    async for chunk in llm.chat_stream(self.messages, temperature=0.9):
        content_parts.append(chunk)
        yield chunk
    
    full_content = "".join(content_parts)
    self.messages.append({"role": "assistant", "content": full_content})
```

**改动点**：
- 文件：`state_machine.py`
- 新增状态常量：`DEBATE`
- 新增方法：`_check_debate_trigger()`, `_start_debate()`
- 在 `handle_user_input()` 中调用辩论检测

```python
def _set_learning_target(self) -> dict:
    """
    根据模式和时长，动态设定学习目标
    """
    from config import MODE_CONTRACT_DEFAULTS, READING_MODE_CONFIG
    
    reading_mode = self.course_info.get("reading_mode", "standard")
    mode_defaults = MODE_CONTRACT_DEFAULTS.get(reading_mode, MODE_CONTRACT_DEFAULTS["standard"])
    
    if reading_mode == "speed":
        # 速读：目标是掌握最重要的N个知识点
        remaining_items = [s for s in self.syllabus_items 
                           if s["status"] != "mastered"]
        target_count = min(5, len(remaining_items))  # 最多5个核心知识点
    elif reading_mode == "standard":
        # 细读：目标是完成当前章节的所有掌握项
        remaining_items = [s for s in self.syllabus_items
                           if s["chapter_index"] == self.chapter_index
                           and s["status"] != "mastered"]
        target_count = len(remaining_items)
    else:
        # 研读：目标是深度理解+辩证分析
        target_count = "unlimited"  # 不设上限，直到完成辩证分析
    
    return {
        "target_count": target_count,
        "goal": mode_defaults["goal"],
        "expected_output": mode_defaults["expected_output"],
    }
```

    for ch in chapters:
        ch_items = [s for s in syllabus if s["chapter_index"] == ch["idx"]]
        chapter_stats.append({
            "title": ch["title"],
            "total": len(ch_items),
            "mastered": sum(1 for s in ch_items if s["status"] == "mastered"),
            "is_loaded": ch.get("is_loaded", 0),
        })
    
    # 阅读模式对应的学习策略
    reading_mode = course.get("reading_mode", "standard")
    strategy = {
        "speed": "精华提炼模式 — 聚焦最重要的20%知识点",
        "standard": "系统学习模式 — 逐章覆盖全部知识点",
        "deep": "辩证分析模式 — 深度理解+批判性思考",
    }
    
    return {
        "reading_mode": reading_mode,
        "strategy_description": strategy[reading_mode],
        "knowledge_coverage": {
            "total_points": total_points,
            "mastered_points": mastered_points,
            "percent": round(mastered_points / max(total_points, 1) * 100, 1),
        },
        "chapter_stats": chapter_stats,
        "weak_areas": _identify_weak_areas(syllabus),
        "recommendation": _generate_recommendation(course, syllabus, reading_mode),
    }


def _identify_weak_areas(syllabus: list) -> list:
    """识别薄弱环节"""
    pending = [s for s in syllabus if s["status"] == "pending"]
    return pending[:5]  # 返回前5个待学习项


def _generate_recommendation(course, syllabus, reading_mode: str) -> str:
    """生成下一步行动建议"""
    progress = db.get_syllabus_progress(course["id"])

    elif progress["percent"] < 100:

### 11.3 关键设计决策

1. **速读模式采用双轨制**：每章快照（fast tier）+ 全局提炼（balanced tier），平衡成本和效果
2. **细读采用均匀采样**：解决前半本书权重过高的问题
3. **研读采用结构化辩证框架**：通过系统提示词强制跨章对比，而非依赖LLM自由发挥
4. **模式-契约联动仅为默认值**：保留用户手动调整的灵活性
5. **所有新功能向后兼容**：已有课程不受影响，新课程默认使用新行为
